import { NextRequest, NextResponse } from 'next/server';
import connectDB from '@/lib/mongodb';
import MatrixEntry from '@/models/MatrixEntry';
import MCQBank from '@/models/MCQBank';
import TestResult from '@/models/TestResult';
import User from '@/models/User';

export async function GET(request: NextRequest) {
  try {
    await connectDB();
    const currentDate = new Date();
    const currentMonth = currentDate.getMonth() + 1; // 1-12
    const currentYear = currentDate.getFullYear();

    // 1. Fetch all Matrix Entries
    const matrixEntries = await MatrixEntry.find().lean();

    // 2. Map Users by name (normalised) → _id
    //    Also support employeeId if present in both MatrixEntry and User
    const distinctNames = [...new Set(matrixEntries.map(e => e.employeeName?.trim()))].filter(Boolean);
    const users = await User.find({ name: { $in: distinctNames } })
      .select('_id name employeeId')
      .lean() as Array<{ _id: any; name: string; employeeId?: string }>;

    const userToIdMap: Record<string, string> = {};
    for (const u of users) {
      userToIdMap[u.name.toLowerCase().trim()] = u._id.toString();
    }

    // 3. Fetch all TestResults for those users and build a completion map
    //    Key: `${userId}_${sopIdentifier}` → best result info
    const userIds = Object.values(userToIdMap);
    const results = await TestResult.find({ userId: { $in: userIds } })
      .select('userId isPassed completedAt score questions')
      .lean() as Array<{ userId: any; isPassed: boolean; completedAt: Date; score: number; questions: any[] }>;

    const completedExamsMap: Record<string, { completedAt: Date; isPassed: boolean; score: number }> = {};
    for (const r of results) {
      const uId = r.userId.toString();
      if (!r.questions?.length) continue;

      const sopIds = [
        ...new Set(
          r.questions
            .map((q: any) => q.sopIdentifier?.trim())
            .filter(Boolean)
        ),
      ] as string[];

      for (const sopId of sopIds) {
        const key = `${uId}_${sopId}`;
        const existing = completedExamsMap[key];
        // Prefer a passing result; amongst equal statuses prefer higher score
        if (!existing || (!existing.isPassed && r.isPassed) || (existing.isPassed === r.isPassed && r.score > existing.score)) {
          completedExamsMap[key] = {
            completedAt: r.completedAt,
            isPassed: r.isPassed,
            score: r.score,
          };
        }
      }
    }

    // 4. SOP metadata from MCQBank — build prefix match lookup once
    const sopsFromBank = await MCQBank.find()
      .select('sopIdentifier sopName totalQuestions')
      .lean() as Array<{ sopIdentifier: string; sopName: string; totalQuestions?: number }>;

    const getSopInfo = (matrixCode: string) => {
      const cleanCode = matrixCode.trim().toUpperCase();
      const matched = sopsFromBank.find((b) => {
        const cleanBankId = b.sopIdentifier.trim().toUpperCase();
        return cleanBankId === cleanCode || cleanBankId.startsWith(`${cleanCode}-`);
      });
      if (matched) {
        return {
          id: matched.sopIdentifier.trim(),
          name: matched.sopName.trim(),
          totalQuestions: matched.totalQuestions ?? 0,
          hasBank: true,
        };
      }
      return { id: matrixCode.trim(), name: 'Unknown SOP', totalQuestions: 0, hasBank: false };
    };

    // 5. Categorize into 4 buckets
    const overdueExams:   any[] = [];
    const dueThisMonth:   any[] = [];
    const upcomingExams:  any[] = [];
    const completedExams: any[] = [];

    for (const entry of matrixEntries) {
      const uId = userToIdMap[entry.employeeName?.toLowerCase().trim() ?? ''];
      const sopInfo = getSopInfo(entry.sopCode);

      // Look up result by full id first, then raw matrix code as fallback
      let resultInfo =
        (uId && completedExamsMap[`${uId}_${sopInfo.id}`]) ||
        (uId && completedExamsMap[`${uId}_${entry.sopCode.trim()}`]) ||
        null;

      const scheduledData = {
        _id: entry._id.toString(),
        trainerName: entry.employeeName,
        department: entry.department,
        sopName: sopInfo.name,
        sopCode: sopInfo.id,
        matrixMonth: entry.monthName,
        matrixMonthIndex: entry.month,
        matrixYear: entry.year,
        hasBank: sopInfo.hasBank,
        totalBankQuestions: sopInfo.totalQuestions,
        hasUserLink: !!uId,
      };

      if (resultInfo) {
        completedExams.push({
          ...scheduledData,
          completedAt: resultInfo.completedAt,
          score: resultInfo.score,
          status: resultInfo.isPassed ? 'Passed' : 'Failed',
        });
        continue;
      }

      // Categorise pending exams by time position
      const isCurrentMonth = entry.year === currentYear && entry.month === currentMonth;
      const isPast = entry.year < currentYear || (entry.year === currentYear && entry.month < currentMonth);

      if (isCurrentMonth) {
        dueThisMonth.push({ ...scheduledData, urgency: 'due' });
      } else if (isPast) {
        overdueExams.push({ ...scheduledData, urgency: 'overdue' });
      } else {
        upcomingExams.push(scheduledData);
      }
    }

    // 6. Warnings: matrix entries with no matching user account
    const unmappedEmployees = [...new Set(
      matrixEntries
        .filter(e => !userToIdMap[e.employeeName?.toLowerCase().trim() ?? ''])
        .map(e => e.employeeName)
    )];

    return NextResponse.json({
      success: true,
      todayExams: dueThisMonth,       // backward-compat key
      dueThisMonth,
      overdueExams,
      completedExams,
      upcomingExams,
      meta: {
        totalMatrix: matrixEntries.length,
        unmappedEmployees,
        generatedAt: new Date().toISOString(),
      },
    });

  } catch (error: any) {
    console.error('API Error in exam-schedules:', error);
    return NextResponse.json({ success: false, error: error.message }, { status: 500 });
  }
}
