
        {/* Comparison Modal */}
        {comparisonMode && selectedQuestion && mostSimilarQuestion && (
          <div className="fixed inset-0 bg-black/70 flex items-center justify-center p-4 z-50">
            <div className="bg-slate-900 rounded-2xl max-w-7xl w-full max-h-[90vh] overflow-y-auto border border-purple-500/30 shadow-[0_0_50px_rgba(0,0,0,0.5)]">
              <div className="sticky top-0 bg-gradient-to-r from-purple-600 to-indigo-600 p-6 rounded-t-2xl z-10">
                <div className="flex items-center justify-between">
                  <div>
                    <h2 className="text-2xl font-bold text-white mb-2">
                      Similar Question Comparison
                    </h2>
                    <p className="text-purple-100 text-sm">
                      {formatSOPDisplayName(selectedQuestion.sopName, selectedQuestion.sopIdentifier)}
                    </p>
                    <div className="mt-2 flex items-center gap-2">
                      <Percent className="h-5 w-5 text-yellow-300" />
                      <span className="text-yellow-300 font-bold text-lg">
                        {mostSimilarQuestion.similarityScore}% Similar
                      </span>
                    </div>
                  </div>
                  <button
                    onClick={() => {
                      setComparisonMode(false);
                      setMostSimilarQuestion(null);
                      setSelectedQuestion(null);
                    }}
                    className="text-white hover:text-gray-200 text-2xl font-bold"
                  >
                    ×
                  </button>
                </div>
              </div>

              <div className="p-6">
                <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                  {/* Primary Question */}
                  <div className="bg-gradient-to-br from-orange-900/20 to-red-900/20 rounded-xl p-6 border border-orange-500/30">
                    <div className="flex items-center gap-3 mb-4">
                      <div className="px-3 py-1 bg-orange-500/20 text-orange-300 rounded-lg font-bold text-sm border border-orange-500/30">
                        Question #{selectedQuestion.primaryQuestion.questionIndex + 1}
                      </div>
                      <div className="px-3 py-1 bg-blue-500/20 text-blue-300 rounded-lg font-bold text-sm border border-blue-500/30">
                        PRIMARY
                      </div>
                    </div>

                    <div className="space-y-4">
                      <div className="flex items-start gap-3">
                        <span className="text-3xl">{selectedQuestion.primaryQuestion.question.aiIcon}</span>
                        <div className="flex-1">
                          <p className="text-white font-semibold text-lg leading-relaxed">
                            {selectedQuestion.primaryQuestion.question.question}
                          </p>
                        </div>
                      </div>

                      <div className="space-y-2">
                        <p className="text-gray-400 text-sm font-bold uppercase tracking-wider">Options:</p>
                        {selectedQuestion.primaryQuestion.question.options.map((option, idx) => (
                          <div
                            key={idx}
                            className={`p-3 rounded-lg border ${
                              option === selectedQuestion.primaryQuestion.question.correctAnswer
                                ? 'bg-green-500/10 border-green-500/30 text-green-300'
                                : 'bg-white/5 border-white/10 text-gray-300'
                            }`}
                          >
                            <span className="font-medium">{String.fromCharCode(65 + idx)}. </span>
                            {option}
                            {option === selectedQuestion.primaryQuestion.question.correctAnswer && (
                              <span className="ml-2 text-green-400 font-bold">✓ Correct</span>
                            )}
                          </div>
                        ))}
                      </div>

                      <div className="bg-white/5 rounded-lg p-4 border border-white/10">
                        <p className="text-gray-400 text-xs font-bold uppercase tracking-wider mb-2">Difficulty:</p>
                        <div className="flex items-center gap-2">
                          <span className="text-yellow-400 text-lg">{selectedQuestion.primaryQuestion.question.difficultyStars}</span>
                          <span className="text-gray-300 font-medium">{selectedQuestion.primaryQuestion.question.difficulty}</span>
                        </div>
                      </div>

                      {selectedQuestion.primaryQuestion.question.explanation && (
                        <div className="bg-blue-900/20 rounded-lg p-4 border border-blue-500/20">
                          <p className="text-blue-300 text-xs font-bold uppercase tracking-wider mb-2">Explanation:</p>
                          <p className="text-gray-300 text-sm leading-relaxed">
                            {selectedQuestion.primaryQuestion.question.explanation}
                          </p>
                        </div>
                      )}
                    </div>
                  </div>

                  {/* Most Similar Question */}
                  <div className="bg-gradient-to-br from-purple-900/20 to-indigo-900/20 rounded-xl p-6 border border-purple-500/30">
                    <div className="flex items-center gap-3 mb-4">
                      <div className="px-3 py-1 bg-purple-500/20 text-purple-300 rounded-lg font-bold text-sm border border-purple-500/30">
                        Question #{mostSimilarQuestion.questionIndex + 1}
                      </div>
                      <div className="px-3 py-1 bg-yellow-500/20 text-yellow-300 rounded-lg font-bold text-sm border border-yellow-500/30">
                        SIMILAR
                      </div>
                    </div>

                    <div className="space-y-4">
                      <div className="flex items-start gap-3">
                        <span className="text-3xl">{mostSimilarQuestion.question.aiIcon}</span>
                        <div className="flex-1">
                          <p className="text-white font-semibold text-lg leading-relaxed">
                            {mostSimilarQuestion.question.question}
                          </p>
                        </div>
                      </div>

                      <div className="space-y-2">
                        <p className="text-gray-400 text-sm font-bold uppercase tracking-wider">Options:</p>
                        {mostSimilarQuestion.question.options.map((option, idx) => (
                          <div
                            key={idx}
                            className={`p-3 rounded-lg border ${
                              option === mostSimilarQuestion.question.correctAnswer
                                ? 'bg-green-500/10 border-green-500/30 text-green-300'
                                : 'bg-white/5 border-white/10 text-gray-300'
                            }`}
                          >
                            <span className="font-medium">{String.fromCharCode(65 + idx)}. </span>
                            {option}
                            {option === mostSimilarQuestion.question.correctAnswer && (
                              <span className="ml-2 text-green-400 font-bold">✓ Correct</span>
                            )}
                          </div>
                        ))}
                      </div>

                      <div className="bg-white/5 rounded-lg p-4 border border-white/10">
                        <p className="text-gray-400 text-xs font-bold uppercase tracking-wider mb-2">Difficulty:</p>
                        <div className="flex items-center gap-2">
                          <span className="text-yellow-400 text-lg">{mostSimilarQuestion.question.difficultyStars}</span>
                          <span className="text-gray-300 font-medium">{mostSimilarQuestion.question.difficulty}</span>
                        </div>
                      </div>

                      {mostSimilarQuestion.question.explanation && (
                        <div className="bg-blue-900/20 rounded-lg p-4 border border-blue-500/20">
                          <p className="text-blue-300 text-xs font-bold uppercase tracking-wider mb-2">Explanation:</p>
                          <p className="text-gray-300 text-sm leading-relaxed">
                            {mostSimilarQuestion.question.explanation}
                          </p>
                        </div>
                      )}

                      {mostSimilarQuestion.matchingText && (
                        <div className="bg-yellow-900/20 rounded-lg p-4 border border-yellow-500/20">
                          <p className="text-yellow-300 text-xs font-bold uppercase tracking-wider mb-2">Matching Keywords:</p>
                          <p className="text-gray-300 text-sm">
                            {mostSimilarQuestion.matchingText}
                          </p>
                        </div>
                      )}
                    </div>
                  </div>
                </div>

                {/* Decision Buttons */}
                <div className="mt-6 bg-white/5 rounded-xl p-6 border border-white/10">
                  <h3 className="text-white font-bold text-lg mb-4">Make Your Decision</h3>
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <button
                      onClick={() => {
                        setComparisonMode(false);
                        setSelectedQuestion(selectedQuestion);
                        setReviewMode(true);
                        setSelectedAction('keep_primary');
                      }}
                      className="px-6 py-4 bg-gradient-to-r from-blue-600 to-cyan-600 hover:from-blue-700 hover:to-cyan-700 text-white font-bold rounded-xl transition-all flex items-center justify-center gap-2"
                    >
                      <CheckCircle2 className="h-5 w-5" />
                      Keep Primary (Q#{selectedQuestion.primaryQuestion.questionIndex + 1})
                    </button>
                    <button
                      onClick={() => {
                        setComparisonMode(false);
                        setSelectedQuestion(selectedQuestion);
                        setReviewMode(true);
                        setSelectedAction('keep_similar');
                        setSelectedQuestionIndex(mostSimilarQuestion.questionIndex);
                      }}
                      className="px-6 py-4 bg-gradient-to-r from-purple-600 to-pink-600 hover:from-purple-700 hover:to-pink-700 text-white font-bold rounded-xl transition-all flex items-center justify-center gap-2"
                    >
                      <CheckCircle2 className="h-5 w-5" />
                      Keep Similar (Q#{mostSimilarQuestion.questionIndex + 1})
                    </button>
                    <button
                      onClick={() => {
                        setComparisonMode(false);
                        setSelectedQuestion(selectedQuestion);
                        setReviewMode(true);
                        setSelectedAction('merge');
                      }}
                      className="px-6 py-4 bg-gradient-to-r from-green-600 to-emerald-600 hover:from-green-700 hover:to-emerald-700 text-white font-bold rounded-xl transition-all flex items-center justify-center gap-2"
                    >
                      <GitMerge className="h-5 w-5" />
                      Merge Both Questions
                    </button>
                    <button
                      onClick={() => {
                        setComparisonMode(false);
                        setSelectedQuestion(selectedQuestion);
                        setReviewMode(true);
                        setSelectedAction('eliminate_all');
                      }}
                      className="px-6 py-4 bg-gradient-to-r from-red-600 to-orange-600 hover:from-red-700 hover:to-orange-700 text-white font-bold rounded-xl transition-all flex items-center justify-center gap-2"
                    >
                      <XCircle className="h-5 w-5" />
                      Eliminate Both
                    </button>
                  </div>
                  <button
                    onClick={() => {
                      setComparisonMode(false);
                      setMostSimilarQuestion(null);
                    }}
                    className="mt-4 w-full px-6 py-3 bg-gray-600 hover:bg-gray-700 text-white font-bold rounded-xl transition-all flex items-center justify-center gap-2"
                  >
                    <X className="h-5 w-5" />
                    Close Comparison
                  </button>
                </div>
              </div>
            </div>
          </div>
        )}
